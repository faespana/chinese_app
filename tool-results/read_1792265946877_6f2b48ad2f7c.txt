   355→function newMatch(){
   356→  const d=deck();
   357→  if(d.length<3){ $("#mtBox").innerHTML='<div class="empty">Necesitas al menos 3 palabras en el mazo filtrado para jugar.</div>'; return; }
   358→  const pick=shuffle(d).slice(0,5);
   359→  if(S.match&&S.match.timer)clearInterval(S.match.timer);
   360→  S.match={pairs:pick, selL:null, selR:null, done:0, t0:null, timer:null};
   361→  renderMatch();
   362→}
   363→function renderMatch(){
   364→  const m=S.match;
   365→  const L=shuffle(m.pairs), R=shuffle(m.pairs);
   366→  $("#mtBox").innerHTML='<div class="mhead"><span>Empareja 汉字 con su significado</span><span>⏱ <span class="mtimer" id="mtT">0.0 s</span></span></div>'+
   367→    '<div class="mrow"><div>'+L.map(w=>'<button class="mcell zh cjk" data-l="'+w.id+'">'+esc(w.chinese)+'</button>').join("")+'</div>'+
   368→    '<div>'+R.map(w=>'<button class="mcell" data-r="'+w.id+'">'+esc(w.spanish)+'</button>').join("")+'</div></div>';
   369→  m.timer=setInterval(()=>{ if(m.t0)document.getElementById("mtT").textContent=((Date.now()-m.t0)/1000).toFixed(1)+" s"; },100);
   370→  document.querySelectorAll("[data-l]").forEach(b=>b.onclick=()=>{
   371→    if(b.classList.contains("ok"))return;
   372→    document.querySelectorAll("[data-l]").forEach(x=>x.classList.remove("sel"));
   373→    b.classList.add("sel"); m.selL=+b.dataset.l; if(!m.t0)m.t0=Date.now(); speak(wById(m.selL).chinese); tryMatch();
   374→  });
   375→  document.querySelectorAll("[data-r]").forEach(b=>b.onclick=()=>{
   376→    if(b.classList.contains("ok"))return;
   377→    document.querySelectorAll("[data-r]").forEach(x=>x.classList.remove("sel"));
   378→    b.classList.add("sel"); m.selR=+b.dataset.r; if(!m.t0)m.t0=Date.now(); tryMatch();
   379→  });
   380→  function tryMatch(){
   381→    if(m.selL&&m.selR){
   382→      const lb=document.querySelector('[data-l="'+m.selL+'"]'), rb=document.querySelector('[data-r="'+m.selR+'"]');
   383→      if(m.selL===m.selR){
   384→        lb.classList.remove("sel"); rb.classList.remove("sel");
   385→        lb.classList.add("ok"); rb.classList.add("ok"); m.done++;
   386→        if(m.done===m.pairs.length){
   387→          clearInterval(m.timer);
   388→          const t=((Date.now()-m.t0)/1000).toFixed(1);
   389→          setTimeout(()=>{ document.getElementById("mtBox").insertAdjacentHTML("beforeend",
   390→            '<div class="final" style="margin-top:20px"><div style="font-size:15px;color:var(--muted)">¡Ronda completa! 🎉</div>'+
   391→            '<div class="score" style="font-size:38px">'+t+' s</div>'+
   392→            '<div class="fcrow"><button class="btn" id="mtNew">Nueva ronda 🔀</button></div></div>');
   393→            document.getElementById("mtNew").onclick=newMatch; },350);
   394→        }
   395→      }else{
   396→        lb.classList.add("bad"); rb.classList.add("bad");
   397→        setTimeout(()=>{ lb.classList.remove("bad","sel"); rb.classList.remove("bad","sel"); },450);
   398→      }
   399→      m.selL=null; m.selR=null;
   400→    }
   401→  }
   402→}
